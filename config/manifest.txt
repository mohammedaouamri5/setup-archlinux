# Deployment manifest.  Read by install.sh (stage configs) and by
# scripts/capture-configs.sh, so capturing and deploying can never drift.
#
#   submodule   <name>   <destination relative to $HOME>
#   link-dir    <path>                       symlink whole directory
#   link-file   <path>                       symlink one static file
#   copy-file   <path>                       real copy (app rewrites by rename)
#   render-file <path>                       copy, replacing REPO_HOME with $HOME
#   render-tmpl <template> <destination>     copy, replacing @HOME@ with $HOME
#
# Paths are relative to config/ on the source side and to $HOME on the target.
# Empty/absent sources are skipped with a warning, so a partial capture is safe.
